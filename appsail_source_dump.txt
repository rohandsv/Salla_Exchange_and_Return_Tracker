    14	  const isProd = nodeEnv === "production" || catalystEnv === "production";

    19	    path.resolve(process.cwd(), ".env"),

    33	  tenantId: string,
    34	  orderNumber: string,

    64	export function encryptText(plain: string): string {
    65	  const key = getEncKey();
    66	  const iv = crypto.randomBytes(12);
    67	  const cipher = crypto.createCipheriv("aes-256-gcm", key, iv);
    68	
    69	  const enc = Buffer.concat([cipher.update(plain, "utf8"), cipher.final()]);
    70	  const tag = cipher.getAuthTag();

    75	export function decryptText(payload: string): string {
    76	  const [v, ivB64, tagB64, dataB64] = payload.split(":");
    77	  if (v !== "v1" || !ivB64 || !tagB64 || !dataB64) throw new Error("Invalid encrypted payload");
    78	
    79	  const key = getEncKey();
    80	  const iv = Buffer.from(ivB64, "base64url");
    81	  const tag = Buffer.from(tagB64, "base64url");
    82	  const data = Buffer.from(dataB64, "base64url");
    83	
    84	  const decipher = crypto.createDecipheriv("aes-256-gcm", key, iv);
    85	  decipher.setAuthTag(tag);
    86	
    87	  const dec = Buffer.concat([decipher.update(data), decipher.final()]);
    88	  return dec.toString("utf8");

     5	export function checkRateLimit(key: string, windowMs: number, max: number): { allowed: boolean; retryAfterSec?: number } {
     6	  const now = Date.now();
     7	  const e = bucket.get(key);
     8	
     9	  if (!e || e.resetAt <= now) {
    10	    bucket.set(key, { count: 1, resetAt: now + windowMs });
    11	    return { allowed: true };
    12	  }
    13	
    14	  if (e.count >= max) {
    15	    const retryAfterSec = Math.ceil((e.resetAt - now) / 1000);
    16	    return { allowed: false, retryAfterSec };
    17	  }
    18	
    19	  e.count += 1;
    20	  bucket.set(key, e);
    21	  return { allowed: true };
    22	}

     9	 * Parse Catalyst DateTime: "YYYY-MM-DD HH:MM:SS"
    10	 * Convert to epoch ms safely without relying on Date(string) quirks.
    11	 */
    12	function catalystDateTimeToMs(dt?: string | null): number {
    13	  if (!dt) return 0;
    14	
    15	  const m = /^(\d{4})-(\d{2})-(\d{2}) (\d{2}):(\d{2}):(\d{2})$/.exec(String(dt).trim());
    16	  if (!m) return 0;
    17	
    18	  const [, y, mo, d, h, mi, s] = m;
    19	
    20	  // Use UTC to be deterministic (only relative interval checks)
    21	  return Date.UTC(
    22	    Number(y),
    23	    Number(mo) - 1,
    24	    Number(d),
    25	    Number(h),
    26	    Number(mi),
    27	    Number(s)
    28	  );
    29	}

   134	      signature_valid: Boolean(payload.signature_valid),

   137	      process_status: "pending",

   139	    };

   156	      return { row: existing, inserted: false };
   157	    }

   118	function sanitizeRelativeReturnTo(v?: string): string | null {
   119	  if (!v) return null;

   123	  if (s.includes("://")) return null;
   124	  if (!/^[\/a-zA-Z0-9\-._~%!$&'()*+,;=:@?]*$/.test(s)) return null;
   125	  return s;
   126	}

   297	  static async getValidAccessTokenForTenant(req: any, tenantId: string) {
   298	    const { clientId, clientSecret, tokenUrl } = getOAuthCoreConfig();
   299	
   300	    const tid = assertDigits(tenantId);
   301	    const row = await SallaOauthTokensRepo.findByTenantId(req, tid);
   302	
   303	    const enc = row?.access_token_enc ? String(row.access_token_enc) : "";
   304	
   305	    // ✅ IMPORTANT: prevent decrypting placeholders like __pending_authorize__
   306	    if (!row || !enc || enc === "__revoked__" || enc.startsWith("__")) {
   307	      throw new AppError(409, "Salla not connected yet", "SALLA_NOT_CONNECTED_YET");
   308	    }
   309	
   310	    const status = String(row.token_status || "").toLowerCase();
   311	
   312	    // If token is active and not near expiry -> use it
   313	    if (status === "active" && !shouldRefresh(row.access_token_expires_at)) {
   314	      return decryptText(enc);
   315	    }
   316	
   317	    const refreshEnc = row.refresh_token_enc ? String(row.refresh_token_enc) : "";
   318	    if (!refreshEnc || refreshEnc === "__revoked__" || refreshEnc.startsWith("__")) {
   319	      await SallaOauthTokensRepo.upsertByTenant(req, tid, { token_status: "missing_refresh_token" });
   320	      throw new AppError(409, "Refresh token missing; reconnect required", "REFRESH_TOKEN_MISSING");
   321	    }
   322	
   323	    const refreshToken = decryptText(refreshEnc);
   324	
   325	    const form = new URLSearchParams({
   326	      grant_type: "refresh_token",
   327	      client_id: clientId,
   328	      client_secret: clientSecret,
   329	      refresh_token: refreshToken,
   330	    });
   331	
   332	    const { ok, status: httpStatus, text, json } = await httpForm(tokenUrl, form);
   333	
   334	    if (!ok) {
   335	      await SallaOauthTokensRepo.upsertByTenant(req, tid, { token_status: "refresh_failed" });
   336	      throw new AppError(502, "Token refresh failed", "OAUTH_REFRESH_FAILED", `HTTP ${httpStatus}: ${text}`);
   337	    }
   338	
   339	    const newAccess = String(json.access_token || "").trim();
   340	    const newRefresh = String(json.refresh_token || "").trim();
   341	    const expiresIn = Number(json.expires_in || 0);
   342	
   343	    if (!newAccess) {
   344	      await SallaOauthTokensRepo.upsertByTenant(req, tid, { token_status: "refresh_failed" });
   345	      throw new AppError(502, "Refresh missing access_token", "OAUTH_REFRESH_INVALID");
   346	    }
   347	
   348	    const nowStr = toCatalystDateTime(new Date());
   349	    const newExpiresAt = toExpiresAt(expiresIn);
   350	
   351	    await SallaOauthTokensRepo.upsertByTenant(req, tid, {
   352	      token_status: "active",
   353	      access_token_enc: encryptText(newAccess),
   354	      refresh_token_enc: newRefresh ? encryptText(newRefresh) : row.refresh_token_enc ?? null,
   355	      access_token_expires_at: newExpiresAt,
   356	      last_token_refresh_at: nowStr,
   357	    });
   358	
   359	    return newAccess;
   360	  }
   361	}

    97	/**
    98	 * Deterministic lifecycle handler.
    99	 * - install/update => ensure tenant by portal slug (if present)
   100	 * - authorize => MUST resolve tenant (storeId/portalSlug), but if portalSlug exists we auto-ensure
   101	 */
   102	export async function handleSallaWebhook(req: any, tenantIdOrNull: string | null, event: SallaWebhookEvent) {
   103	  const t = normalizeEventType(event);
   104	  const storeId = extractMerchantId(event);
   105	  const portalSlug = extractPortalSlug(event);
   106	
   107	  // 1) uninstall
   108	  if (isUninstallEvent(t)) {
   109	    const tid = await resolveTenantId(req, tenantIdOrNull, storeId, portalSlug);
   110	    if (!tid) return { ok: true, handled: "uninstall", tenant: null };
   111	
   112	    const nowStr = toCatalystDateTime(new Date());
   113	    await SallaOauthTokensRepo.revoke(req, tid, nowStr);
   114	    await TenantsRepo.updateSallaConnectionFields(req, tid, { status: "uninstalled" });
   115	
   116	    return { ok: true, handled: "uninstall", tenant: tid };
   117	  }
   118	
   119	  // 2) install/update => ensure tenant by portal slug (so authorize later can link)
   120	  if (isInstallEvent(t) || isUpdateEvent(t)) {
   121	    if (!portalSlug) return { ok: true, handled: "ignored", type: t || null };
   122	
   123	    const tenant = await TenantsRepo.ensureTenantByPortalSlug(req, portalSlug, {
   124	      status: "installed_pending_authorize",
   125	      plan_code: "free",
   126	    });
   127	
   128	    // if storeId is present, attach it (helps future routing)
   129	    if (storeId) {
   130	      await TenantsRepo.updateSallaConnectionFields(req, tenant.ROWID, {
   131	        salla_store_id: storeId,
   132	        status: "installed_pending_authorize",
   133	      });
   134	    } else {
   135	      await TenantsRepo.updateSallaConnectionFields(req, tenant.ROWID, { status: "installed_pending_authorize" });
   136	    }
   137	
   138	    return { ok: true, handled: isInstallEvent(t) ? "install" : "update", tenant: tenant.ROWID, store_id: storeId ?? null };
   139	  }
   140	
   141	  // 3) authorize
   142	  if (isAuthorizeEvent(t)) {
   143	    if (!storeId) throw new AppError(400, "Missing merchant id in authorize event", "AUTHORIZE_MISSING_MERCHANT");
   144	
   145	    // If portalSlug exists, ensure tenant (install might not have been called manually)
   146	    if (portalSlug) {
   147	      await TenantsRepo.ensureTenantByPortalSlug(req, portalSlug, { status: "installed_pending_authorize", plan_code: "free" }).catch(() => null);
   148	    }
   149	
   150	    const tid = await resolveTenantId(req, tenantIdOrNull, storeId, portalSlug);
   151	    if (!tid) throw new AppError(404, "Tenant not resolved for authorize event", "TENANT_NOT_RESOLVED");
   152	
   153	    const tokens = extractTokens(event);
   154	    if (!tokens) throw new AppError(400, "Missing tokens in authorize event", "AUTHORIZE_MISSING_TOKENS");
   155	
   156	    const nowStr = toCatalystDateTime(new Date());
   157	    const expiresAt = toExpiresAt(tokens.expires_in);
   158	
   159	    const existing = await SallaOauthTokensRepo.findByTenantId(req, tid).catch(() => null);
   160	
   161	    await SallaOauthTokensRepo.upsertByTenant(req, tid, {
   162	      tenant_unique_key: existing?.tenant_unique_key ? String(existing.tenant_unique_key) : `store:${storeId}`,
   163	      token_status: "active",
   164	      access_token_enc: encryptText(tokens.access_token),
   165	      refresh_token_enc: tokens.refresh_token ? encryptText(tokens.refresh_token) : existing?.refresh_token_enc ?? null,
   166	      token_type: tokens.token_type || "Bearer",
   167	      scopes: tokens.scope,

   193	export async function processPendingSallaWebhooks(req: any, opts: { limit?: number; maxRetries?: number } = {}) {
   194	  const limit = Number.isFinite(opts.limit) && (opts.limit as number) > 0 ? (opts.limit as number) : 25;
   195	  const maxRetries = Number.isFinite(opts.maxRetries) && (opts.maxRetries as number) >= 0 ? (opts.maxRetries as number) : 5;
   196	
   197	  const rows = await WebhookEventsRepo.listPendingBatch(req, { limit, maxRetries });
   198	
   199	  let done = 0;
   200	  let failed = 0;
   201	  let ignored = 0;
   202	
   203	  for (const row of rows) {
   204	    await WebhookEventsRepo.markProcessing(req, row.ROWID).catch(() => null);
   205	
   206	    if (!row.signature_valid) {
   207	      await WebhookEventsRepo.markDone(req, row.ROWID).catch(() => null);
   208	      ignored++;
   209	      continue;
   210	    }
   211	
   212	    const eventObj = safeJsonParse<SallaWebhookEvent>(row.payload_json, {} as any);
   213	    const tenantIdOrNull = row.tenant_id == null ? null : String(row.tenant_id);
   214	
   215	    try {
   216	      await handleSallaWebhook(req, tenantIdOrNull, eventObj);
   217	      await WebhookEventsRepo.markDone(req, row.ROWID).catch(() => null);
   218	      done++;
   219	    } catch {
   220	      await WebhookEventsRepo.markFailedAndIncrementRetry(req, row.ROWID, row.retry_count).catch(() => null);
   221	      failed++;
   222	    }
   223	  }
   224	
   225	  return { ok: true, scanned: rows.length, done, failed, ignored };
   226	}
